Overview
Trendin LLM Query Builder generates 100+ natural, brand-free search queries optimized for LLM visibility.

Key Features
95% generic queries (no brand names) + 5% branded queries (target brand only)
Region-specific analysis (US, UK, UAE, India, Switzerland, etc.)
Web crawling with auto-translation for non-English pages
Industry tagging to ensure relevant content
Google News integration via SERP API
Knowledge Graph enrichment via Google KG API
Twitter/X trends integration (optional)
Output
100+ unique search queries with intent classification
CSV export: query, topic, intent, country_iso_code
Architecture
Tech Stack
Frontend: React 18 + TypeScript + Vite + Tailwind CSS + shadcn/ui
Backend:  Express.js + TypeScript
AI:       OpenAI API (gpt-4o-mini)
APIs:     SERP API, Google Knowledge Graph API, Twitter/X API v2

10-Step Pipeline
Pipeline Flow (server/routes.ts)
async function processAnalysis(id: string) {
  const analysis = await storage.getAnalysis(id);
  const region = analysis.region || 'us';
  const data: any = {};
  // Step 0: Web Crawl
  const crawledData = await aiService.crawlWebsite(analysis.domain);
  
  // Step 1: Domain Analysis
  data.domainAnalysis = await aiService.generateDomainAnalysis(
    analysis.domain, crawledData, region
  );
  // Step 2: Entity Extraction + Knowledge Graph Enrichment
  data.entities = await aiService.generateEntities(
    analysis.domain, data.domainAnalysis, crawledData, region
  );
  // Optional: Enrich with Google Knowledge Graph
  if (kgApiKey) {
    const { enrichedEntities, triples } = await enrichEntitiesWithKnowledgeGraph(
      kgApiKey, data.entities
    );
    data.kgTriples = triples;
  }
  // Step 3: Semantic Expansion
  data.expand = await aiService.generateExpansion(
    data.entities, analysis.domain, crawledData, data.kgTriples
  );
  // Step 4: Knowledge Graph
  data.graph = await aiService.generateGraph(data.entities, data.expand);
  // Step 5: Pain Points
  data.painPoints = await aiService.generatePainPoints(
    analysis.domain, data.entities, crawledData, region, data.domainAnalysis
  );
  // Step 6: Jobs-to-be-Done
  data.jtbd = await aiService.generateJTBD(
    analysis.domain, data.painPoints, data.entities, data.domainAnalysis, region
  );
  // Step 7: Goals Framework
  data.goals = await aiService.generateGoals(
    analysis.domain, data.jtbd, data.domainAnalysis, region
  );
  // Step 8: Personas
  data.personas = await aiService.generatePersonas(
    analysis.domain, data.painPoints, data.entities, data.jtbd, region
  );
  // Step 9: Trending Insights (Google News + Twitter)
  data.trending = await generateTrendingInsights(
    data.entities, data.painPoints, data.jtbd, twitterToken,
    analysis.domain, personasList, domainContext, region, brandName
  );
  // Step 10: Query Generation
  // 10a. Generate 95% generic queries
  const aiQueries = await aiService.generateQueries(
    analysis.domain, data.personas, data.painPoints, 
    data.entities, data.trending, data.goals, data.domainAnalysis
  );
  
  // 10b. Add news-based queries
  const newsQueries = data.trending.newsBasedQueries;
  
  // 10c. Deduplicate
  let uniqueQueries = deduplicateQueries([...newsQueries, ...aiQueries]);
  
  // 10d. Generate supplementary queries if < 100
  if (uniqueQueries.length < 100) {
    const supplementary = await aiService.generateSupplementaryQueries(...);
    uniqueQueries = deduplicateQueries([...uniqueQueries, ...supplementary]);
  }
  
  // 10e. Filter out branded queries
  const filtered = uniqueQueries.filter(q => !containsBrand(q.query, brandBlacklist));
  
  // 10f. Add 5% branded queries
  const brandedQueries = await aiService.generateBrandedQueries(
    analysis.domain, productCategory, Math.ceil(filtered.length * 0.05)
  );
  
  data.queries = [...filtered, ...brandedQueries];
}

// Normalize domain URL
function normalizeDomain(domain: string): string {
  let normalized = domain.trim().toLowerCase();
  if (!normalized.startsWith("http://") && !normalized.startsWith("https://")) {
    normalized = "https://" + normalized;
  }
  return normalized;
}

// Detect if content is English
function isEnglishContent(text: string): boolean {
  const englishWords = ['the', 'and', 'is', 'in', 'to', 'of', 'for', 'with'];
  const lowerText = text.toLowerCase();
  const words = lowerText.split(/\s+/);
  let englishWordCount = 0;
  for (const word of words) {
    if (englishWords.includes(word)) englishWordCount++;
  }
  return (englishWordCount / words.length) >= 0.05;
}

    hl: hl,       // Language code (always "en")
    google_domain: google_domain,  // Regional Google (google.ch, google.co.uk)
  });

    .filter(article => {

      return dateStr.includes("hour") || dateStr.includes("day") || 
             dateStr.includes("yesterday") || dateStr.includes("week");
    })
    .slice(0, 15);
}

Brand Filtering Logic
const KNOWN_BRAND_BLACKLIST = [
  // VPN brands
  'expressvpn', 'nordvpn', 'surfshark', 'cyberghost', 'privatevpn',
  'protonvpn', 'ipvanish', 'purevpn', 'tunnelbear', 'windscribe',
  // Tech brands
  'google', 'amazon', 'microsoft', 'apple', 'facebook', 'meta',
  'netflix', 'disney', 'spotify', 'dropbox', 'slack', 'zoom',
  // ... more brands
];
function buildBrandBlacklist(domain: string, entities: any[], domainAnalysis?: any): string[] {
  const blacklist = new Set<string>();
  
  // Add domain-derived brand name
  const brandName = domain.replace(/^(https?:\/\/)?(www\.)?/, '')
    .split('.')[0].split('/')[0].toLowerCase();
  blacklist.add(brandName);
  
  // Add known brands
  KNOWN_BRAND_BLACKLIST.forEach(b => blacklist.add(b));
  
  // Extract competitors from entities
  for (const entity of entities) {
    if (entity.category?.toLowerCase().includes('competitor')) {
      entity.items.forEach((item: string) => blacklist.add(item.toLowerCase()));
    }
  }
  
  // Extract competitors from domain analysis
  if (domainAnalysis?.competitors) {
    domainAnalysis.competitors.forEach((c: string) => blacklist.add(c.toLowerCase()));
  }
  
  return Array.from(blacklist);
}
function containsBrand(query: string, blacklist: string[]): boolean {
  const queryLower = query.toLowerCase();
  return blacklist.some(brand => {
    const brandPattern = new RegExp(`\\b${brand}\\b`, 'i');
    return brandPattern.test(queryLower);
  });
}

app.get("/api/analyses/:id/export", async (req, res) => {
  const analysis = await storage.getAnalysis(req.params.id);
  const queries = analysis.data.queries;
  const region = analysis.region || "us";
  const topic = analysis.data.domainAnalysis?.industry || "general";
  
  const normalizeIntent = (intent: string): string => {
    const map: Record<string, string> = {
      "informational": "Informational",
      "commercial": "Commercial",
      "transactional": "Transactional",
      "navigational": "Navigational"
    };
    return map[intent.toLowerCase()] || "Informational";
  };
  
  const csv = [
    "query,topic,intent,country_iso_code",
    ...queries.map((q: any) => 
      `${q.query},${topic},${normalizeIntent(q.intent)},${region}`
    )
  ].join("\n");
  res.setHeader("Content-Type", "text/csv");
  res.setHeader("Content-Disposition", `attachment; filename="queries.csv"`);
  res.send(csv);
});

Key Design Decisions
Domain NOT passed to query prompt: Prevents brand contamination in queries
Industry tagging: Every step includes industry context for relevance
Region-specific analysis: Competitors, personas, and trends are region-specific
English-only output: All regions use hl=en; non-English pages auto-translated
95/5 split: 95% generic queries + 5% branded navigational queries
Deduplication: Queries normalized and deduplicated
Supplementary generation: If <100 queries, additional queries generated
